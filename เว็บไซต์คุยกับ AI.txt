def fetch_github_responses(url):
    try:
        response = requests.get(url, timeout=5)

            return response.json()
    except Exception as e:
        print(f"Error fetching GitHub JSON: {e}")
    return None

@app.route("/")
def index():
    if "user" in session:
        return redirect(url_for("chat"))
    return render_template_string(INDEX_HTML)


@app.route("/register", methods=["GET", "POST"])
def register():
    error = None
    if request.method == "POST":
        username = request.form.get("username", "").strip()
        password = request.form.get("password", "").strip()
        confirm_password = request.form.get("confirm_password", "").strip()


        if not re.match("^[A-Za-z]+$", username):
            error = "ชื่อต้องเป็นภาษาอังกฤษเท่านั้น (A-Z, a-z)"
        elif username in users_db:
            error = "ชื่อนี้ถูกใช้งานแล้ว กรุณาเลือกชื่ออื่น"
        elif password in FORBIDDEN_PASSWORDS:
            error = "ห้ามใช้รหัสผ่านง่ายเกินไป เช่น 12345678, 87654321, 21436587"
        elif password != confirm_password:
            error = "รหัสผ่านและยืนยันรหัสผ่านไม่ตรงกัน"
        else:
            users_db[username] = password
            chat_history_db[username] = []
            return redirect(url_for("login"))


    return render_template_string(REGISTER_HTML, error=error)


@app.route("/login", methods=["GET", "POST"])
def login():
    error = None
    if request.method == "POST":
        username = request.form.get("username", "").strip()
        password = request.form.get("password", "").strip()


        if username in users_db and users_db[username] == password:
            session["user"] = username
            return redirect(url_for("chat"))
        else:
            error = "ชื่อผู้ใช้หรือรหัสผ่านไม่ถูกต้อง"


    return render_template_string(LOGIN_HTML, error=error)


@app.route("/logout")
def logout():
    session.pop("user", None)
    return redirect(url_for("index"))


@app.route("/chat")
def chat():
    if "user" not in session:
        return redirect(url_for("login"))
    username = session["user"]
    history = chat_history_db.get(username, [])
    return render_template_string(CHAT_HTML, username=username, history=history)


@app.route("/send_message", methods=["POST"])
def send_message():
    if "user" not in session:
        return jsonify({"error": "Unauthorized"}), 401


    username = session["user"]
    user_message = request.json.get("message", "").strip()
    if not user_message:
        return jsonify({"error": "Empty message"})


    data = fetch_github_responses(DEFAULT_GITHUB_URL)
    bot_response = "ขออภัยครับ ระบบไม่พบคำตอบที่คุณต้องการในขณะนี้"


    if data and isinstance(data, dict):
        user_text_lower = user_message.lower()
        collected_responses = []


        for key_word, options in data.items():
            if key_word.lower() == "no delta":
                continue
            if key_word.lower() in user_text_lower:
                if isinstance(options, list) and options:
                    collected_responses.append(random.choice(options))
        
        if collected_responses:
            bot_response = " ".join(collected_responses)
        else:
            if "No Delta" in data and isinstance(data["No Delta"], list) and data["No Delta"]:
                bot_response = random.choice(data["No Delta"])


    chat_history_db[username].append({"sender": "user", "text": user_message})
    chat_history_db[username].append({"sender": "bot", "text": bot_response})


    return jsonify({"response": bot_response})
